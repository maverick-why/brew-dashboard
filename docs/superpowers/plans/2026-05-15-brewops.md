```sql
-- 酒厂
CREATE TABLE breweries (
  id         uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  name       text NOT NULL,
  slug       text UNIQUE NOT NULL,
  created_at timestamptz DEFAULT now()
);

-- 成员
CREATE TABLE brewery_members (
  id           uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  brewery_id   uuid REFERENCES breweries NOT NULL,
  user_id      uuid REFERENCES auth.users NOT NULL,
  role         text CHECK (role IN ('owner','brewer')) NOT NULL,
  display_name text NOT NULL,
  status       text CHECK (status IN ('active','pending')) DEFAULT 'pending',
  created_at   timestamptz DEFAULT now(),
  UNIQUE(brewery_id, user_id)
);

-- 批次
CREATE TABLE batches (
  id          uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  brewery_id  uuid REFERENCES breweries NOT NULL,
  tank        text NOT NULL,
  beer        text NOT NULL,
  batch_no    text,
  color       text NOT NULL DEFAULT '#1D9E75',
  og          text,
  yeast       text,
  notes       text,
  plan        jsonb NOT NULL,
  created_at  timestamptz DEFAULT now(),
  archived_at timestamptz
);

-- 操作日志
CREATE TABLE batch_logs (
  id         uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  brewery_id uuid REFERENCES breweries NOT NULL,
  batch_id   uuid REFERENCES batches NOT NULL,
  step_key   text NOT NULL,
  operator   text,
  ts         timestamptz NOT NULL DEFAULT now(),
  temp       text,
  brix       text,
  ph         text,
  note       text,
  is_defer   boolean DEFAULT false
);

-- RLS 辅助函数
CREATE FUNCTION my_brewery_id() RETURNS uuid AS $$
  SELECT brewery_id FROM brewery_members
  WHERE user_id = auth.uid() AND status = 'active' LIMIT 1;
$$ LANGUAGE sql STABLE SECURITY DEFINER;

CREATE FUNCTION my_role() RETURNS text AS $$
  SELECT role FROM brewery_members
  WHERE user_id = auth.uid() AND status = 'active' LIMIT 1;
$$ LANGUAGE sql STABLE SECURITY DEFINER;

-- RLS
ALTER TABLE breweries ENABLE ROW LEVEL SECURITY;
CREATE POLICY "read own brewery" ON breweries
  FOR SELECT USING (id = my_brewery_id());

ALTER TABLE brewery_members ENABLE ROW LEVEL SECURITY;
CREATE POLICY "read members" ON brewery_members
  FOR SELECT USING (brewery_id = my_brewery_id());
CREATE POLICY "owner manage" ON brewery_members
  FOR ALL USING (brewery_id = my_brewery_id() AND my_role() = 'owner');

ALTER TABLE batches ENABLE ROW LEVEL SECURITY;
CREATE POLICY "tenant" ON batches FOR ALL USING (brewery_id = my_brewery_id());

ALTER TABLE batch_logs ENABLE ROW LEVEL SECURITY;
CREATE POLICY "tenant" ON batch_logs FOR ALL USING (brewery_id = my_brewery_id());

-- 邀请激活 trigger
CREATE FUNCTION handle_user_confirmed() RETURNS trigger AS $$
BEGIN
  UPDATE brewery_members SET status = 'active'
  WHERE user_id = NEW.id AND status = 'pending';
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE TRIGGER on_auth_user_confirmed
  AFTER UPDATE OF email_confirmed_at ON auth.users
  FOR EACH ROW WHEN (NEW.email_confirmed_at IS NOT NULL)
  EXECUTE FUNCTION handle_user_confirmed();
```

```js
export const tod = () => { const d = new Date(); d.setHours(0,0,0,0); return d }
export const addD = (d, n) => { const r = new Date(d); r.setDate(r.getDate()+n); return r }
export const diffD = (a, b) => Math.round((new Date(b) - new Date(a)) / 86400000)
export const fmtD = d => { if(!d) return '—'; const x=new Date(d); return `${x.getMonth()+1}/${x.getDate()}` }
export const fmtFull = d => { if(!d) return ''; const x=new Date(d); return `${x.getFullYear()}-${String(x.getMonth()+1).padStart(2,'0')}-${String(x.getDate()).padStart(2,'0')}` }
export const fmtTs = ts => { const d=new Date(ts); return `${d.getMonth()+1}/${d.getDate()} ${String(d.getHours()).padStart(2,'0')}:${String(d.getMinutes()).padStart(2,'0')}` }
export const parseD = s => { if(!s) return null; const d=new Date(s); d.setHours(0,0,0,0); return isNaN(d)?null:d }

export function stepStatus(dateVal) {
  if (!dateVal) return 'future'
  const diff = diffD(tod(), new Date(dateVal))
  if (diff < 0)  return 'done'
  if (diff === 0) return 'today'
  if (diff === 1) return 'tomorrow'
  if (diff <= 3)  return 'soon'
  return 'future'
}
```

```js
import { createContext, useContext, useEffect, useState } from 'react'
import { supabase } from '../lib/supabase.js'

const AuthCtx = createContext(null)

export function AuthProvider({ children }) {
  const [session, setSession] = useState(undefined) // undefined = loading

  useEffect(() => {
    supabase.auth.getSession().then(({ data }) => setSession(data.session))
    const { data: { subscription } } = supabase.auth.onAuthStateChange((_, s) => setSession(s))
    return () => subscription.unsubscribe()
  }, [])

  async function signIn(email, password) {
    const { error } = await supabase.auth.signInWithPassword({ email, password })
    if (error) throw error
  }

  async function signOut() {
    await supabase.auth.signOut()
  }

  async function resetPassword(email) {
    const { error } = await supabase.auth.resetPasswordForEmail(email, {
      redirectTo: `${window.location.origin}/brewops/reset-password`,
    })
    if (error) throw error
  }

  return (
    <AuthCtx.Provider value={{ session, user: session?.user ?? null, loading: session === undefined, signIn, signOut, resetPassword }}>
      {children}
    </AuthCtx.Provider>
  )
}

export const useAuth = () => useContext(AuthCtx)
```

```js
import { createContext, useContext, useEffect, useState } from 'react'
import { supabase } from '../lib/supabase.js'
import { useAuth } from './useAuth.js'

const BreweryCtx = createContext(null)

export function BreweryProvider({ children }) {
  const { user } = useAuth()
  const [brewery, setBrewery]   = useState(null)
  const [role, setRole]         = useState(null)
  const [members, setMembers]   = useState([])
  const [loading, setLoading]   = useState(true)

  useEffect(() => {
    if (!user) { setBrewery(null); setRole(null); setLoading(false); return }
    load()
  }, [user])

  async function load() {
    setLoading(true)
    const { data: mem } = await supabase
      .from('brewery_members')
      .select('role, brewery_id, display_name, breweries(id, name, slug)')
      .eq('user_id', user.id)
      .eq('status', 'active')
      .single()

    if (!mem) { setBrewery(null); setRole(null); setLoading(false); return }

    setRole(mem.role)
    setBrewery(mem.breweries)

    const { data: all } = await supabase
      .from('brewery_members')
      .select('id, display_name, role, status, user_id')
    setMembers(all ?? [])
    setLoading(false)
  }

  async function createBrewery(name, slug, displayName) {
    const { data: b, error: be } = await supabase
      .from('breweries')
      .insert({ name, slug })
      .select()
      .single()
    if (be) throw be

    const { error: me } = await supabase
      .from('brewery_members')
      .insert({ brewery_id: b.id, user_id: user.id, role: 'owner', display_name: displayName, status: 'active' })
    if (me) throw me

    await load()
  }

  async function removeMember(memberId) {
    const { error } = await supabase.from('brewery_members').delete().eq('id', memberId)
    if (error) throw error
    setMembers(prev => prev.filter(m => m.id !== memberId))
  }

  return (
    <BreweryCtx.Provider value={{ brewery, role, members, loading, createBrewery, removeMember, reload: load }}>
      {children}
    </BreweryCtx.Provider>
  )
}

export const useBrewery = () => useContext(BreweryCtx)
```

```js
    setLoading(false)
  }

  // Convert plan jsonb string dates → Date objects, logs ts → numbers
  function normalize(rows) {
    return rows.map(b => ({
      ...b,
      plan: Object.fromEntries(
        Object.entries(b.plan).map(([k, v]) => [k, v ? new Date(v) : null])
      ),
      logs: (b.batch_logs ?? []).map(l => ({ ...l, ts: new Date(l.ts).getTime() })),
    }))
  }

  async function addBatch({ tank, beer, batch_no, color, og, yeast, notes, plan }) {
    const planStr = Object.fromEntries(
      Object.entries(plan).map(([k, v]) => [k, v instanceof Date ? v.toISOString().slice(0,10) : v])
    )
    const { data, error } = await supabase
      .from('batches')
      .insert({ brewery_id: breweryId, tank, beer, batch_no, color, og, yeast, notes, plan: planStr })
      .select('*, batch_logs(*)')
      .single()
    if (error) throw error
    setBatches(prev => [...prev, ...normalize([data])])
  }

  async function updateBatch(id, fields) {
    const update = { ...fields }
    if (update.plan) {
      update.plan = Object.fromEntries(
        Object.entries(update.plan).map(([k, v]) => [k, v instanceof Date ? v.toISOString().slice(0,10) : v])
      )
    }
    const { error } = await supabase.from('batches').update(update).eq('id', id)
    if (error) throw error
    setBatches(prev => prev.map(b => b.id === id ? { ...b, ...fields } : b))
  }

  async function archiveBatch(id) {
    const { error } = await supabase.from('batches').update({ archived_at: new Date().toISOString() }).eq('id', id)
    if (error) throw error
    setBatches(prev => prev.filter(b => b.id !== id))
  }

  async function saveLog({ batchId, stepKey, operator, ts, temp, brix, ph, note, newDate, isDefer }) {
    const row = {
      brewery_id: breweryId, batch_id: batchId,
      step_key: stepKey, operator, ts: new Date(ts).toISOString(),
      temp, brix, ph, note, is_defer: isDefer ?? false,
    }
    // upsert by batch_id + step_key (replace existing non-defer log)
    const { data, error } = isDefer
      ? await supabase.from('batch_logs').insert(row).select().single()
      : await supabase.from('batch_logs').upsert(row, { onConflict: 'batch_id,step_key', ignoreDuplicates: false }).select().single()
    if (error) throw error

    if (newDate) await updateBatch(batchId, { plan: { ...(batches.find(b=>b.id===batchId)?.plan ?? {}), [stepKey]: newDate } })

    setBatches(prev => prev.map(b => {
      if (b.id !== batchId) return b
      const logs = isDefer
        ? [...b.logs, { ...data, ts: new Date(data.ts).getTime() }]
        : [...b.logs.filter(l => l.step_key !== stepKey), { ...data, ts: new Date(data.ts).getTime() }]
      return { ...b, logs }
    }))
  }

  async function deferStep({ batchId, stepKey, newDate, note }) {
    await saveLog({ batchId, stepKey: `_defer_${stepKey}`, operator: '', ts: Date.now(), note, isDefer: true })
    await updateBatch(batchId, {
      plan: { ...(batches.find(b=>b.id===batchId)?.plan ?? {}), [stepKey]: newDate }
    })
  }

  return { batches, loading, addBatch, updateBatch, archiveBatch, saveLog, deferStep }
}
```

```js
module.exports = async (req, res) => {
  if (req.method !== 'POST') return res.status(405).end()

  // Verify caller is authenticated + is owner
  const token = req.headers.authorization?.replace('Bearer ', '')
  if (!token) return res.status(401).json({ error: 'Unauthorized' })

  const supabase = getSupabase()
  const { data: { user }, error: authErr } = await supabase.auth.getUser(token)
  if (authErr || !user) return res.status(401).json({ error: 'Invalid token' })

  const { data: member } = await supabase
    .from('brewery_members')
    .select('role, brewery_id')
    .eq('user_id', user.id)
    .eq('status', 'active')
    .single()

  if (!member || member.role !== 'owner') return res.status(403).json({ error: 'Owner only' })

  const { email } = req.body
  if (!email) return res.status(400).json({ error: 'email required' })

  const admin = getAdminClient()

  // Send invite email via Supabase
  const { error: inviteErr } = await admin.auth.admin.inviteUserByEmail(email, {
    data: { brewery_id: member.brewery_id, role: 'brewer' },
  })
  if (inviteErr) return res.status(500).json({ error: inviteErr.message })

  // Pre-create pending member row
  await supabase.from('brewery_members').upsert({
    brewery_id: member.brewery_id,
    user_id: (await admin.auth.admin.listUsers()).data.users.find(u => u.email === email)?.id ?? '00000000-0000-0000-0000-000000000000',
    role: 'brewer',
    display_name: email.split('@')[0],
    status: 'pending',
  }, { onConflict: 'brewery_id,user_id', ignoreDuplicates: true })

  return res.status(200).json({ ok: true })
}
```

```jsx
    })
    return { today, tomorrow, soon }
  }

  const { today: todayT, tomorrow: tomT, soon: soonT } = getTasks()
  const urgentN = [...todayT, ...tomT].filter(t => !batches.find(b => b.id === t.batch.id)?.logs.some(l => l.step_key === t.step.k)).length

  function openLog(batch, step) {
    const ex = batch.logs.find(l => l.step_key === step.k) ?? null
    setLogForm({ batch, step, op: ex?.operator ?? (OPS[0] ?? ''), note: ex?.note ?? '',
      temp: ex?.temp ?? '', brix: ex?.brix ?? '', ph: ex?.ph ?? '' })
  }

  if (loading) return <Spinner />

  return (
    <div style={{ background:C.bg, minHeight:'100%', fontFamily:'system-ui,-apple-system,sans-serif', color:C.tx, maxWidth:460, margin:'0 auto', position:'relative' }}>
      {/* Header */}
      <div style={{ padding:'18px 16px 0', background:C.bg, position:'sticky', top:0, zIndex:10 }}>
        <div style={{ display:'flex', alignItems:'center', gap:8, marginBottom:14 }}>
          <div style={{ flex:1 }}>
            <div style={{ fontSize:10, color:C.mu, letterSpacing:'0.1em', textTransform:'uppercase', marginBottom:2 }}>{brewery?.name}</div>
            <div style={{ fontSize:20, fontWeight:700, color:C.tx }}>发酵管理</div>
          </div>
          {urgentN > 0 && (
            <span onClick={() => setView('today')} style={{ display:'inline-flex', alignItems:'center', padding:'3px 10px', borderRadius:12, background:C.red, color:'#fff', fontSize:12, fontWeight:700, cursor:'pointer' }}>
              {urgentN} 待处理
            </span>
          )}
          <button onClick={() => navigate('/brewops/settings')}
            style={{ background:'none', border:`1px solid ${C.bd}`, borderRadius:20, padding:'7px 12px', color:C.mu, fontSize:12 }}>
            设置
          </button>
          <button onClick={() => setAddModal(true)}
            style={{ background:C.fa, border:`1px solid ${C.g}44`, borderRadius:20, padding:'7px 14px', color:C.dg, fontSize:13, fontWeight:600 }}>
            + 新增
          </button>
        </div>
        <div style={{ display:'flex', borderBottom:`1px solid ${C.bd}` }}>
          {[['today', `今日任务${urgentN > 0 ? ` (${urgentN})` : ''}`], ['gantt','总览'], ['logs','记录']].map(([t, lbl]) => (
            <button key={t} onClick={() => setView(t)} style={{
              flex:1, padding:'11px 0', fontSize:13, fontWeight:600, border:'none', background:'none', cursor:'pointer',
              color: view === t ? C.tx : C.mu,
              borderBottom: view === t ? `2px solid ${C.g}` : '2px solid transparent',
            }}>{lbl}</button>
          ))}
        </div>
      </div>

      <div style={{ overflowY:'auto', maxHeight:520, paddingBottom:20 }}>
        {view === 'today' && <TodayView batches={batches} todayT={todayT} tomT={tomT} soonT={soonT} onLog={openLog} onEdit={b => setEditBatch({ ...b })} />}
        {view === 'gantt' && <GanttView batches={batches} ganttStart={ganttStart} ganttDays={ganttDays} setGanttStart={setGanttStart} setGanttDays={setGanttDays} onLog={openLog} onEdit={b => setEditBatch({ ...b })} />}
        {view === 'logs'  && <LogsView batches={batches} />}
      </div>

      {logForm   && <LogSheet   logForm={logForm} ops={OPS} onSave={async d => { await saveLog({ batchId: logForm.batch.id, ...d }); setLogForm(null) }} onClose={() => setLogForm(null)} />}
      {editBatch && <EditSheet  batch={editBatch} onSave={async u => { await updateBatch(u.id, u); setEditBatch(null) }} onDelete={async id => { await archiveBatch(id); setEditBatch(null) }} onLog={openLog} onClose={() => setEditBatch(null)} />}
      {addModal  && <AddSheet   onAdd={async d => { await addBatch(d); setAddModal(false) }} onClose={() => setAddModal(false)} count={batches.length} />}
    </div>
  )
}

export default function App() {
  return (
    <Routes>
      <Route path="/brewops/login"    element={<LoginPage />} />
      <Route path="/brewops/setup"    element={<RequireAuth><SetupPage /></RequireAuth>} />
      <Route path="/brewops/settings" element={<RequireOwner><SettingsPage /></RequireOwner>} />
      <Route path="/brewops/*"        element={<RequireAuth><MainApp /></RequireAuth>} />
    </Routes>
  )
}
```
